cmake_minimum_required(VERSION 3.22)
project(rntrace CXX)

# librntrace: the process's one tracer (include/rntrace/rntrace.h). Shared, so every library linking it (the app's
# RnTrace TurboModule, react-native-libav, ...) records into the same instance.
#   RNTRACE_ENABLED=ON  the Perfetto backend, when its SDK is there (scripts/fetch-perfetto.sh)
#   RNTRACE_ENABLED=OFF a no-op library with the same API (release builds by default)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(RNTRACE_ENABLED "Build the Perfetto backend" ON)
set(PERFETTO_SDK_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/perfetto/sdk)

add_library(rntrace SHARED src/RnTrace.cpp src/RnTraceJni.cpp)
target_include_directories(rntrace PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_definitions(rntrace PRIVATE RNTRACE_BUILDING)
target_compile_options(rntrace PRIVATE -fvisibility=hidden)
if(ANDROID)
  target_link_options(rntrace PRIVATE -Wl,-z,max-page-size=16384) # 16 KB page sizes (Android 15+ devices)
endif()

if(RNTRACE_ENABLED AND EXISTS ${PERFETTO_SDK_DIR}/perfetto.cc)
  target_sources(rntrace PRIVATE ${PERFETTO_SDK_DIR}/perfetto.cc)
  set_source_files_properties(${PERFETTO_SDK_DIR}/perfetto.cc PROPERTIES COMPILE_OPTIONS "-w")
  target_include_directories(rntrace PRIVATE ${PERFETTO_SDK_DIR})
  target_compile_definitions(rntrace PRIVATE RNTRACE_PERFETTO)
  if(ANDROID)
    target_link_libraries(rntrace PRIVATE log)
  endif()
  message(STATUS "rn-trace: Perfetto backend (${PERFETTO_SDK_DIR})")
elseif(RNTRACE_ENABLED)
  message(STATUS "rn-trace: no Perfetto SDK (scripts/fetch-perfetto.sh): no-op backend")
else()
  message(STATUS "rn-trace: no-op backend (RNTRACE_ENABLED=OFF)")
endif()
